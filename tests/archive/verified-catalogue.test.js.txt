const {test}=require('node:test');
const assert=require('node:assert/strict');
const fs=require('node:fs');
const crypto=require('node:crypto');
const catalog=require('../data/catalog.json');
const audit=require('../data/price-audit.json');
const {buildCatalog}=require('../lib/catalog');
const {createOrder}=require('../lib/orders');
const {makeServer}=require('../server');
const {choices,tierCount}=require('../lib/tier-options');
const ctx=buildCatalog(catalog);
const personalisation=(p,v)=>({colouring:'natural',allergens:'accept',tiers:tierCount(p,v)>1?Array.from({length:tierCount(p,v)},(_,i)=>({tier:i+1,type:i===0?'edible':'dummy',sponge:i===0?choices.sponges[0]:'',filling:i===0?choices.fillings[0]:''})):[]});
const payload=(p,v)=>({items:[{productId:p.id,variantId:v.id,quantity:2,personalisation:personalisation(p,v)}],customer:{name:'Catalogue Test',phone:'+971500000000',address:'Dubai',date:'2099-12-01'},consent:true});

test('every active design has its own original US or Canadian photograph and no generated image',()=>{
 const hashes=new Set(),sourcePhotos=new Set(),titles=new Set();
 for(const p of catalog.products){
  assert.ok(['US','CA'].includes(p.source.country));assert.ok(!titles.has(p.title));titles.add(p.title);
  assert.ok(!sourcePhotos.has(p.source.originalImage));sourcePhotos.add(p.source.originalImage);
  const bytes=fs.readFileSync(p.source.originalFile),hash=crypto.createHash('sha256').update(bytes).digest('hex');assert.ok(!hashes.has(hash),p.title);hashes.add(hash);
  assert.ok(p.image.startsWith('assets/catalogue/'));assert.ok(fs.existsSync(p.image));
  assert.deepEqual(p.images,[p.image]);assert.ok(p.variants.every(v=>v.image===p.image));
  assert.equal(ctx.byId.get(p.id).title,p.title);assert.equal(ctx.byId.get(p.id).image,p.image);
 }
 assert.ok(catalog.products.some(p=>p.source.country==='CA'));
});

test('all source variants convert to AED then receive exactly AED 300 once',()=>{
 assert.equal(audit.length,catalog.products.flatMap(p=>p.variants).length);
 for(const p of catalog.products){
  for(const v of p.variants){
   const row=audit.find(a=>a.variantId===v.id&&a.productId===p.id);assert.ok(row);
   const sourceName=require('../data/verified-selection.json').find(r=>r.sourceUrl===row.sourceUrl).source;
   const source=require('../data/sources/'+sourceName+'.json').products.find(s=>String(s.id)===p.id);
   assert.equal(row.sourcePrice,Number(source.variants.find(s=>String(s.id)===v.id).price));
   if(p.quoteOnly){assert.equal(v.priceFils,null);assert.equal(row.markupAED,null);continue;}
   const expected=Math.round(Math.round(row.sourcePrice*100)*row.aedPerUnit)+30000;
   assert.equal(v.priceFils,expected);assert.equal(row.sellingPriceFils,expected);
  }
  assert.equal(p.minPriceFils,p.quoteOnly?null:Math.min(...p.variants.map(v=>v.priceFils)));
 }
});

test('tier filters show the price and link for a matching tier variant',()=>{
 const results=ctx.query('wedding-cakes',new URLSearchParams('tier=6-plus')).products.filter(p=>!p.quoteOnly);
 assert.ok(results.length);
 for(const p of results){
  const selected=p.variants.find(v=>v.id===p.listingVariantId);
  assert.ok(selected.tiers>=6,p.title);
  assert.equal(p.listingPriceFils,Math.min(...p.variants.filter(v=>v.tiers>=6).map(v=>v.priceFils)));
 }
});

test('tier choices validate server-side, preserve edible flavours and clear dummy flavours',()=>{
 const p=catalog.products.find(p=>p.tiers===4),v=p.variants[0],input=payload(p,v);
 input.items[0].personalisation.tiers[1].sponge='Untrusted';
 const order=createOrder(input,catalog);assert.equal(order.items[0].personalisation.tiers[1].sponge,'');
 assert.equal(order.items[0].personalisation.tiers[0].sponge,'Vanilla');
 assert.equal(order.subtotalFils,v.priceFils*2);
 for(const tiers of [[],[{type:'dummy'}],Array(4).fill({type:'edible',sponge:'Invalid',filling:'Chocolate ganache'})])assert.throws(()=>createOrder({...input,items:[{...input.items[0],personalisation:{...input.items[0].personalisation,tiers}}]},catalog));
});

test('priced and unpriced wedding quotes keep totals honest, and all public routes render',async t=>{
 const server=makeServer({catalog,config:{name:'Maison Xavi',siteUrl:'https://maison-xavi.example'}});await new Promise(r=>server.listen(0,'127.0.0.1',r));t.after(()=>{server.closeAllConnections();server.close();});const base='http://127.0.0.1:'+server.address().port;
 for(const p of catalog.products){
  const response=await fetch(base+'/cakes/'+p.handle);assert.equal(response.status,200,p.handle);const html=await response.text();assert.ok(!html.includes('assets/studio/'));assert.ok(html.includes(p.image));
  const v=p.variants[0],input=payload(p,v),quote=await (await fetch(base+'/api/quote',{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify(input)})).json();
  const order=createOrder(input,catalog);
  assert.equal(quote.subtotalFils,order.subtotalFils);assert.equal(quote.totalFils,order.totalFils);assert.deepEqual(quote.items[0].personalisation.tiers,order.items[0].personalisation.tiers);
  if(p.quoteOnly){assert.equal(quote.totalFils,null);assert.equal(quote.deliveryFeeFils,10000);assert.equal(order.deliveryFeeFils,10000);assert.ok(html.includes('Price on request'));assert.ok(!html.includes('AED 0'));}
 }
 for(const c of ctx.collections)assert.equal((await fetch(base+'/collections/'+c.slug)).status,200,c.slug);
 for(const route of ['/',...Object.keys(require('../pages/policies').policies),'/photography'])assert.equal((await fetch(base+route)).status,200,route);
 const sitemap=await (await fetch(base+'/sitemap.xml')).text();for(const p of catalog.products)assert.ok(sitemap.includes(p.handle));
 assert.ok(!sitemap.includes('/checkout'));assert.ok(ctx.query('wedding-cakes',new URLSearchParams('tier=6-plus')).total>0);
 assert.ok(ctx.query('fresh-flower-cakes',new URLSearchParams()).products.every(p=>p.flowerType==='fresh'));
 assert.ok(!ctx.query('wedding-cakes',new URLSearchParams('max=100000')).products.some(p=>p.quoteOnly));
});
